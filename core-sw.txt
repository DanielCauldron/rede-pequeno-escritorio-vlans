! ============================================================
! CORE-SW | Switch Layer 3 (3560-24PS) - configuração final
! Projeto: rede-pequeno-escritorio-vlans
! Papel: núcleo da rede. Cria as VLANs, roteia entre elas (SVIs),
!        entrega DHCP e aplica a ACL dos visitantes.
! ============================================================

enable
configure terminal
hostname CORE-SW
ip routing

! --- VLANs ---
vlan 10
 name GERENCIAMENTO
vlan 20
 name EQUIPE
vlan 30
 name SERVIDORES
vlan 40
 name VISITANTES
vlan 50
 name CCTV-IOT
 exit

! --- Trunks para os switches de acesso (3560 exige encapsulation) ---
interface gigabitEthernet 0/1
 description TRUNK-SW-EQUIPE
 switchport trunk encapsulation dot1q
 switchport mode trunk
interface gigabitEthernet 0/2
 description TRUNK-SW-VISITANTES
 switchport trunk encapsulation dot1q
 switchport mode trunk
interface fastEthernet 0/1
 description TRUNK-SW-SERVIDORES
 switchport trunk encapsulation dot1q
 switchport mode trunk
interface fastEthernet 0/2
 description TRUNK-SW-CCTV
 switchport trunk encapsulation dot1q
 switchport mode trunk

! --- Porta roteada para o FW-ROUTER ---
interface fastEthernet 0/3
 description LINK-FW-ROUTER
 no switchport
 ip address 10.0.0.1 255.255.255.252
 no shutdown

! --- ACL: visitantes não acessam a rede interna, mas navegam na internet ---
ip access-list extended BLOQUEIA-VISITANTES
 deny ip 192.168.40.0 0.0.0.255 192.168.10.0 0.0.0.255
 deny ip 192.168.40.0 0.0.0.255 192.168.20.0 0.0.0.255
 deny ip 192.168.40.0 0.0.0.255 192.168.30.0 0.0.0.255
 deny ip 192.168.40.0 0.0.0.255 192.168.50.0 0.0.0.255
 permit ip any any
 exit

! --- SVIs (gateways de cada VLAN) ---
interface vlan 10
 ip address 192.168.10.1 255.255.255.0
 no shutdown
interface vlan 20
 ip address 192.168.20.1 255.255.255.0
 no shutdown
interface vlan 30
 ip address 192.168.30.1 255.255.255.0
 no shutdown
interface vlan 40
 ip address 192.168.40.1 255.255.255.0
 ip access-group BLOQUEIA-VISITANTES in
 no shutdown
interface vlan 50
 ip address 192.168.50.1 255.255.255.0
 no shutdown
 exit

! --- Rota padrão: tudo que não é interno sai pelo FW-ROUTER ---
ip route 0.0.0.0 0.0.0.0 10.0.0.2

! --- DHCP (.1 a .10 reservados para IP fixo) ---
ip dhcp excluded-address 192.168.20.1 192.168.20.10
ip dhcp pool EQUIPE
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
ip dhcp excluded-address 192.168.40.1 192.168.40.10
ip dhcp pool VISITANTES
 network 192.168.40.0 255.255.255.0
 default-router 192.168.40.1

end
write memory

! ============================================================
! VERIFICAÇÃO
!  show vlan brief | show interfaces trunk | show ip interface brief
!  show ip route | show ip dhcp binding | show access-lists
! ============================================================
