! ============================================================
! SWITCHES DE ACESSO (2960)
! Papel: ligar os dispositivos finais. Cada porta pertence a uma VLAN.
! IP de gerenciamento na VLAN 10 (acesso remoto ao switch).
! ============================================================

! ---------------- SW-EQUIPE ----------------
enable
configure terminal
hostname SW-EQUIPE
interface gigabitEthernet 0/1
 description TRUNK-CORE-SW
 switchport mode trunk
interface fastEthernet 0/1
 description PC-EQUIPE
 switchport mode access
 switchport access vlan 20
interface fastEthernet 0/2
 description AP-EQUIPE
 switchport mode access
 switchport access vlan 20
interface vlan 10
 ip address 192.168.10.2 255.255.255.0
 no shutdown
ip default-gateway 192.168.10.1
end
write memory
