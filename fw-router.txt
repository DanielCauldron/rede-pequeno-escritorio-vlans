! ============================================================
! FW-ROUTER (2911) - borda da rede interna + NAT
! Papel: leva o tráfego interno até a internet traduzindo
!        os IPs privados (NAT overload / PAT).
! ============================================================

enable
configure terminal
hostname FW-ROUTER

interface gigabitEthernet 0/0
 description LINK-CORE-SW
 ip address 10.0.0.2 255.255.255.252
 ip nat inside
 no shutdown
interface gigabitEthernet 0/1
 description LINK-ISP
 ip address 200.200.200.1 255.255.255.252
 ip nat outside
 no shutdown
 exit

! Rota padrão para o provedor
ip route 0.0.0.0 0.0.0.0 200.200.200.2

! Rotas de volta para cada VLAN (via CORE-SW)
ip route 192.168.10.0 255.255.255.0 10.0.0.1
ip route 192.168.20.0 255.255.255.0 10.0.0.1
ip route 192.168.30.0 255.255.255.0 10.0.0.1
ip route 192.168.40.0 255.255.255.0 10.0.0.1
ip route 192.168.50.0 255.255.255.0 10.0.0.1

! NAT: quem pode sair (todas as redes 192.168.x.x)
access-list 1 permit 192.168.0.0 0.0.255.255
ip nat inside source list 1 interface gigabitEthernet 0/1 overload

end
write memory
