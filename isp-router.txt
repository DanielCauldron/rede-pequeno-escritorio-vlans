! ============================================================
! ISP-ROUTER - simula o provedor de internet
! Rede 8.8.8.0/24 com o INTERNET-SERVER (8.8.8.8, gw 8.8.8.1)
! ============================================================

enable
configure terminal
hostname ISP-ROUTER

interface gigabitEthernet 0/0
 description LINK-FW-ROUTER
 ip address 200.200.200.2 255.255.255.252
 no shutdown
interface gigabitEthernet 0/1
 description REDE-INTERNET
 ip address 8.8.8.1 255.255.255.0
 no shutdown

end
write memory

! O ISP não precisa de rota para as redes privadas:
! o FW-ROUTER traduz tudo para 200.200.200.1 (NAT).
